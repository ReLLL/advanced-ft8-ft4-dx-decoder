cmake_minimum_required(VERSION 3.20)
project(advanced_ft8_ft4_dx_decoder VERSION 1.0.24 LANGUAGES CXX OBJCXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BROWN_SOURCE "${CMAKE_SOURCE_DIR}/build" CACHE PATH "Pinned Brown ABI source")
set(BROWN_APP "/Applications/SDR++Brown.app" CACHE PATH "Installed Brown")
option(SANITIZE "Address and undefined behavior checks" OFF)
find_package(Threads REQUIRED)
find_package(SQLite3 REQUIRED)
add_library(dx_core STATIC src/model.cpp src/table_options.cpp src/country_summary.cpp src/store.cpp src/helper_client.cpp src/engine.cpp src/cartography.cpp src/csv_log.cpp src/gridtracker.cpp)
set_target_properties(dx_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(dx_core PUBLIC src "${BROWN_SOURCE}/core/src" /opt/homebrew/include)
target_link_libraries(dx_core PUBLIC SQLite3::SQLite3 Threads::Threads)
target_compile_options(dx_core PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(dx_core PRIVATE DX_VERSION="${PROJECT_VERSION}")
if(SANITIZE)
 target_compile_options(dx_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(dx_core PUBLIC -fsanitize=address,undefined)
endif()
set(DECODER "${BROWN_SOURCE}/decoder_modules/ft8_decoder/src")
file(GLOB MSHV CONFIGURE_DEPENDS "${DECODER}/ft8_etc/*.cpp" "${DECODER}/ft8_etc/*.c")
add_library(dx_mshv STATIC ${MSHV} "${DECODER}/fftw_mshv_plug.cpp" "${BROWN_SOURCE}/core/src/utils/flog.cpp")
target_include_directories(dx_mshv SYSTEM PUBLIC "${DECODER}" "${BROWN_SOURCE}/core/src" /opt/homebrew/include)
file(GLOB FFTW "${BROWN_APP}/Contents/Frameworks/libfftw3f.*.dylib")
target_link_libraries(dx_mshv PUBLIC ${FFTW})
target_compile_options(dx_mshv PRIVATE -O3 -Wno-deprecated-declarations -Wno-register)
if(SANITIZE)
 target_compile_options(dx_mshv PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(dx_mshv PUBLIC -fsanitize=address,undefined)
endif()
add_executable(dx_decoder src/helper.cpp)
target_link_libraries(dx_decoder PRIVATE dx_mshv dx_core)
set_target_properties(dx_decoder PROPERTIES BUILD_RPATH "${BROWN_APP}/Contents/Frameworks")
add_custom_command(TARGET dx_decoder POST_BUILD COMMAND install_name_tool -change /opt/homebrew/opt/fftw/lib/libfftw3f.3.dylib @rpath/libfftw3f.3.dylib $<TARGET_FILE:dx_decoder>)
add_library(dx_notebook MODULE src/plugin.cpp src/plugin_ui.cpp src/station_actions.cpp src/help.cpp src/map.cpp src/settings.cpp src/tiles.cpp src/map_links.mm src/corner.cpp src/table_views.cpp src/details.cpp)
target_compile_definitions(dx_notebook PRIVATE DX_VERSION="${PROJECT_VERSION}" DX_VERSION_MAJOR=${PROJECT_VERSION_MAJOR} DX_VERSION_MINOR=${PROJECT_VERSION_MINOR} DX_VERSION_PATCH=${PROJECT_VERSION_PATCH})
file(GLOB VOLK "${BROWN_APP}/Contents/Frameworks/libvolk.*.dylib")
target_include_directories(dx_notebook SYSTEM PRIVATE "${BROWN_SOURCE}/core/src" "${BROWN_SOURCE}/core/src/imgui" /opt/homebrew/include)
target_link_libraries(dx_notebook PRIVATE dx_core "${BROWN_APP}/Contents/Frameworks/libsdrpp_core.dylib" ${VOLK} "-framework AudioToolbox" "-framework OpenGL" "-framework AppKit" "-lcurl")
set_target_properties(dx_notebook PROPERTIES PREFIX "" SUFFIX ".dylib" BUILD_RPATH "${BROWN_APP}/Contents/Frameworks")
enable_testing()
add_executable(dx_tests tests/tests.cpp)
target_link_libraries(dx_tests PRIVATE dx_core)
add_test(NAME dx_tests COMMAND dx_tests)
add_executable(dx_table_columns_tests tests/table_columns.cpp)
target_link_libraries(dx_table_columns_tests PRIVATE dx_core)
add_test(NAME dx_table_columns COMMAND dx_table_columns_tests)
add_executable(dx_country_summary_tests tests/country_summary.cpp)
target_link_libraries(dx_country_summary_tests PRIVATE dx_core)
add_test(NAME dx_country_summary COMMAND dx_country_summary_tests)
add_executable(dx_corner_feed_tests tests/corner_feed.cpp)
target_link_libraries(dx_corner_feed_tests PRIVATE dx_core)
add_test(NAME dx_corner_feed COMMAND dx_corner_feed_tests)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_executable(dx_logging_tests tests/logging.cpp)
target_link_libraries(dx_logging_tests PRIVATE dx_core)
add_test(NAME dx_logging COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_logging.py $<TARGET_FILE:dx_logging_tests>)
add_executable(dx_gridtracker_tests tests/gridtracker.cpp)
target_link_libraries(dx_gridtracker_tests PRIVATE dx_core)
add_test(NAME dx_gridtracker COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_gridtracker.py $<TARGET_FILE:dx_gridtracker_tests>)
add_executable(dx_native_assets tests/native_assets.mm src/map_links.mm)
target_link_libraries(dx_native_assets PRIVATE dx_core "-framework AppKit")
add_test(NAME dx_native_assets COMMAND dx_native_assets)
add_executable(dx_benchmark tests/benchmark.cpp)
target_link_libraries(dx_benchmark PRIVATE dx_core)
add_executable(dx_mock tests/mock_helper.cpp)
target_link_libraries(dx_mock PRIVATE dx_core)
add_executable(dx_recovery tests/recovery.cpp)
target_link_libraries(dx_recovery PRIVATE dx_core)
add_test(NAME dx_recovery COMMAND dx_recovery $<TARGET_FILE:dx_mock>)
add_executable(dx_moderation_tests tests/moderation.cpp)
target_link_libraries(dx_moderation_tests PRIVATE dx_core)
add_test(NAME dx_moderation COMMAND dx_moderation_tests $<TARGET_FILE:dx_mock>)
add_executable(dx_fixture tests/fixture.cpp)
add_executable(dx_visual_fixture tests/visual_fixture.cpp)
target_link_libraries(dx_visual_fixture PRIVATE dx_core)
target_link_libraries(dx_fixture PRIVATE dx_mshv)
add_executable(dx_baseline tests/baseline.cpp "${DECODER}/sdrpp_ft8_mshv.cpp")
target_include_directories(dx_baseline SYSTEM PRIVATE "${BROWN_SOURCE}/core/src/imgui")
target_link_libraries(dx_baseline PRIVATE dx_mshv ${VOLK})
set_target_properties(dx_baseline PROPERTIES BUILD_RPATH "${BROWN_APP}/Contents/Frameworks")
foreach(target dx_notebook dx_baseline)
 add_custom_command(TARGET ${target} POST_BUILD COMMAND install_name_tool -change /usr/local/lib/libvolk.3.3.dylib @rpath/libvolk.3.3.dylib $<TARGET_FILE:${target}>)
endforeach()
